#### 任务表 (task)

```SQL
CREATE TABLE `task` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '任务ID',
    `task_code` VARCHAR(64) NOT NULL DEFAULT '' COMMENT '任务编码',
    `task_name` VARCHAR(256) NOT NULL DEFAULT '' COMMENT '任务名称',
    `task_type` VARCHAR(32) NOT NULL DEFAULT '' COMMENT '任务类型：AI_TAGGING/DATA_STATISTICS/REPORT_SUMMARY',
    `executor` VARCHAR(64) NOT NULL DEFAULT '' COMMENT '主任务执行器：TaggingParentExecutor/StatisticsExecutor/SummaryExecutor',
    
    -- 任务参数
    `task_params` TEXT NOT NULL COMMENT '任务参数（JSON格式）',
    
    -- 状态管理
    `status` VARCHAR(32) NOT NULL DEFAULT 'PENDING' COMMENT '状态：PENDING/RUNNING/SUCCESS/FAILED/CANCELLED',
    
    -- 子任务统计
    `sub_task_count` INT NOT NULL DEFAULT 0 COMMENT '子任务总数',
    `sub_task_success` INT NOT NULL DEFAULT 0 COMMENT '子任务成功数',
    `sub_task_failed` INT NOT NULL DEFAULT 0 COMMENT '子任务失败数',
    `sub_task_processing` INT NOT NULL DEFAULT 0 COMMENT '子任务处理中数',
    
    -- 执行时间
    `execute_time` DATETIME NOT NULL COMMENT '计划执行时间',
    `start_time` DATETIME DEFAULT NULL COMMENT '实际开始时间',
    `end_time` DATETIME DEFAULT NULL COMMENT '实际结束时间',
    
    -- 执行结果
    `result` TEXT COMMENT '执行结果汇总（JSON格式）',
    `error_msg` VARCHAR(2000) NOT NULL DEFAULT '' COMMENT '错误信息',
    
    -- 重试
    `retry_count` INT NOT NULL DEFAULT 0 COMMENT '已重试次数',
    `max_retry` INT NOT NULL DEFAULT 3 COMMENT '最大重试次数',
    
    -- 审计
    `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    `create_by` VARCHAR(64) NOT NULL DEFAULT '' COMMENT '创建人',
    `valid` TINYINT NOT NULL DEFAULT 1 COMMENT '是否生效',
    
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_task_code` (`task_code`),
    KEY `idx_status` (`status`),
    KEY `idx_execute_time` (`execute_time`),
    KEY `idx_task_type` (`task_type`),
    KEY `idx_executor` (`executor`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='任务主表';
```

  

  

####  子任务分析表 (sub_task)

  

```SQL
CREATE TABLE `sub_task` (
    `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '子任务ID',
    `sub_task_code` VARCHAR(64) NOT NULL DEFAULT '' COMMENT '子任务编码',
    `parent_task_id` BIGINT UNSIGNED NOT NULL DEFAULT 0 COMMENT '父任务ID',
    `executor` VARCHAR(64) NOT NULL DEFAULT '' COMMENT '子任务执行器：TaggingChildExecutor/ImageAnalyzeExecutor/VideoAnalyzeExecutor',
    
    -- 业务标识
    `business_type` VARCHAR(32) NOT NULL DEFAULT '' COMMENT '业务类型：MATERIAL_ANALYSIS/DATA_CALC',
    `business_key` VARCHAR(128) NOT NULL DEFAULT '' COMMENT '业务标识（如material_id）',
    
    -- 任务参数
    `sub_task_params` TEXT NOT NULL COMMENT '子任务参数（JSON格式）',
    
    -- 状态管理
    `status` VARCHAR(32) NOT NULL DEFAULT 'PENDING' COMMENT '状态：PENDING/RUNNING/SUCCESS/FAILED/SKIPPED',

    -- 执行进度
    `progress` TEXT COMMENT '执行进度（JSON格式，存储断点续传信息）',

    -- 执行结果
    `result` TEXT COMMENT '子任务执行结果（JSON格式）',
    `error_msg` VARCHAR(2000) NOT NULL DEFAULT '' COMMENT '错误信息',
    
    -- 执行时间
    `execute_time` DATETIME DEFAULT NULL COMMENT '计划执行时间',
    `start_time` DATETIME DEFAULT NULL COMMENT '实际开始时间',
    `end_time` DATETIME DEFAULT NULL COMMENT '实际结束时间',
    `duration_ms` INT NOT NULL DEFAULT 0 COMMENT '耗时毫秒',
    
    -- 优先级和重试
    `priority` INT NOT NULL DEFAULT 5 COMMENT '优先级：1-10',
    `retry_count` INT NOT NULL DEFAULT 0 COMMENT '已重试次数',
    `max_retry` INT NOT NULL DEFAULT 3 COMMENT '最大重试次数',
    
    -- 审计
    `create_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    `update_time` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    `valid` TINYINT NOT NULL DEFAULT 1 COMMENT '是否生效',
    
    PRIMARY KEY (`id`),
    UNIQUE KEY `uk_sub_task_code` (`sub_task_code`),
    KEY `idx_parent_task_id` (`parent_task_id`),
    KEY `idx_status` (`status`),
    KEY `idx_executor` (`executor`),
    KEY `idx_business` (`business_type`, `business_key`),
    KEY `idx_execute_time` (`execute_time`),
    KEY `idx_parent_status` (`parent_task_id`, `status`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='子任务表';
```
